# dsp-citrinet-c2c: the DSP 25 half of the voice assistant.
#
#   I2S mic -> Citrinet-256 CTC transcript -> C2C link -> Bearly ML 25 (bearly-smollm-c2c) -> answer
#
# It is the proven `dsp-citrinet` mic demo compiled with DSP_CITRINET_C2C=1, plus one new source
# file (src/stt_link_dsp.c) that speaks the protocol in c2c-demos/common/stt_link_proto.h. The
# Citrinet sources are REUSED, not copied, so the transcription path here and in dsp-citrinet cannot
# drift apart.
#
#   make build CHIP=dsp25 PLATFORM=CHIP TARGET=dsp-citrinet-c2c EXTRA_CMAKE_ARGS="-DLINKER=llm"
#   make tsi-run TTY=<tty> BINARY=build/c2c-demos/dsp-citrinet-c2c/dsp-citrinet-c2c.elf
#
# Start this chip FIRST (or together with Bearly): it blocks on the boot barrier until Bearly ML 25
# announces that SmolLM is loaded, and it performs no cross-link write before then — writing into a
# still-booting chip kills that chip (see /CLAUDE.md).
#
# LINKER=llm is REQUIRED (the 11.8 MB model lives in .rodata and the encoder wants several MB of
# heap; the stock dsp25.ld gives a 1 MB heap) and the top-level Makefile does not forward LINKER=,
# so it has to go inside EXTRA_CMAKE_ARGS.
#
# ------------------------------------------------------------------------------------------------
# WHY EVERY CACHE VARIABLE BELOW IS PREFIXED `CN_C2C_`
#
# CMake `option()` and `set(... CACHE ...)` create GLOBAL cache entries, and a second declaration of
# a name that already exists is a NO-OP — the first one wins. c2c-demos/CMakeLists.txt adds
# dsp-citrinet before this directory, so reusing its names (CITRINET_USE_RVV, CITRINET_TILE_T, ...)
# silently inherited ITS defaults here: this target shipped with CITRINET_USE_RVV=OFF, i.e. the
# scalar encoder, which is ~3.4x slower (28 s vs 8.2 s on the golden clip) — and nothing in the
# build log said so. Never reuse a sibling demo's cache-variable name.
#
# The knobs that are known-bad are not knobs at all here: they are hardcoded to 0 below, so no cache
# state, stale or fresh, can turn them on. .claude/plans/007-citrinet-dsp.md has the measurements.
# ------------------------------------------------------------------------------------------------

set(_CN_SRC ${CMAKE_CURRENT_SOURCE_DIR}/../dsp-citrinet)
set(_TS_SRC ${CMAKE_SOURCE_DIR}/bearly25-bmarks/tinyspeech-mc)

# The wake gate needs the MFCC front-end. dsp-kws-rolling proved this library on dsp25 silicon.
if(NOT TARGET mfcclib)
  message(FATAL_ERROR "dsp-citrinet-c2c needs target `mfcclib` for the wake gate. "
                      "Configure with -DBUILD_MFCC_LIB=ON.")
endif()

# The wake model's weights. Generated by dsp25-tests/tinyspeech-test/scripts/train_wakeword.py;
# committed, so a normal build needs no Python. Regenerate only to retrain.
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include/wake/weights.h")
  message(FATAL_ERROR "dsp-citrinet-c2c: include/wake/weights.h is missing. Generate it with "
                      "dsp25-tests/tinyspeech-test/scripts/train_wakeword.py")
endif()

add_executable(dsp-citrinet-c2c
  ${_CN_SRC}/src/main.c
  ${_CN_SRC}/src/citrinet.c
  ${_CN_SRC}/src/model_data.S
  src/stt_link_dsp.c
  src/wake_gate.c
  # TinySpeech runtime. Pure C + RVV intrinsics — no hal_ope, nothing bearly-specific — so the same
  # sources the Bearly KWS demos use compile unchanged for dsp25. Not a library target: every demon
  # that uses TinySpeech compiles these directly, because each supplies its own weights.h.
  ${_TS_SRC}/src/tinyspeech_model.c
  ${_TS_SRC}/src/tinyspeech_int8.c
  ${_TS_SRC}/src/tensor.c
  ${_TS_SRC}/src/misc.c
  ${_TS_SRC}/src/modules.c
  ${CMAKE_SOURCE_DIR}/c2c-demos/common/c2c_shm.c   # the one flush / repeat-write implementation
  ${CMAKE_SOURCE_DIR}/bmark-lib/simple_setup.c     # init_test / PLL + UART bring-up
)

# C2C mode implies mic mode: the prompt is whatever was spoken. citrinet_config.h enforces this with
# an #error too, so the pair can never be set inconsistently from the command line.
target_compile_definitions(dsp-citrinet-c2c PRIVATE
  DSP_CITRINET_C2C=1
  DSP_CITRINET_USE_MIC=1
  DSP_CITRINET_FROM_AUDIO=0)

# ---- link geometry (declared once in c2c-demos/CMakeLists.txt — must match the Bearly side) ------
target_compile_definitions(dsp-citrinet-c2c PRIVATE
  STT_LINK_RETURN_ANSWER=${STT_LINK_RETURN_ANSWER_VALUE}
  STT_PROMPT_MAX_BYTES=${STT_PROMPT_MAX_BYTES}u
  STT_ANSWER_MAX_BYTES=${STT_ANSWER_MAX_BYTES}u
  DSP_CITRINET_TRANSCRIPT_MAX=${STT_PROMPT_MAX_BYTES})

# ---- the measured-best encoder configuration, NOT overridable ------------------------------------
# From the optimisation scoreboard in .claude/plans/007-citrinet-dsp.md (silicon, 3 s of audio):
#   RVV kernels ON + activation tiling      28,000 -> 8,239 -> 5,118 ms   <- the two big wins
#   residual MR=4, ReLU fused into the store                              <- always on in citrinet.c
#   MR=6 register blocking      0.98x (more memory streams)     -> OFF
#   activation packing          0.94x (+80 KB working set)      -> OFF
#   DMA weight staging          0.92x, and the strided gather corrupts on silicon -> OFF
#   CONV1D depthwise offload    hangs the build that was measured          -> OFF
#   dual-core                   1.15x but HANGS on rerun                   -> OFF
# These are conclusions, not preferences, so they are compile definitions rather than cache options:
# a stale cache entry cannot silently undo them (which is exactly how this target shipped scalar).
target_compile_definitions(dsp-citrinet-c2c PRIVATE
  CN_USE_RVV=1
  CN_MR6=0
  CN_PACK=0
  CN_CONV1D=0
  CN_DMA_PROBE=0
  CN_DMA_STAGE=0
  CN_DUALCORE=0
  CN_DEBUG_GEOM=0
  CN_TRACE_BLOCK=-1
  CN_DETERMINISM_CHECK=0
  CN_EVICT_CHECK=0
  CN_POISON=0
  CN_MEMTEST_MB=0)

# Activation t-tile. 32 is the measured optimum (16 -> 5,579 ms, 32 -> 5,118, 64 -> 5,138); kept as a
# knob because it is the one parameter worth re-sweeping if the cache or bitstream changes.
set(CN_C2C_TILE_T 32 CACHE STRING "Activation t-tile for the RVV kernels (multiple of VL)")
target_compile_definitions(dsp-citrinet-c2c PRIVATE CN_TILE_T=${CN_C2C_TILE_T})

# Per-kernel cycle profile (frontend / depthwise / pointwise / SE / residual / decoder). Off for the
# demo; turn it on when the transcript latency needs attributing.
option(CN_C2C_PROFILE "Per-kernel cycle profile" OFF)
set(_CN_PROF 0)
if(CN_C2C_PROFILE)
  set(_CN_PROF 1)
  message(STATUS "dsp-citrinet-c2c: CN_C2C_PROFILE=1")
endif()
target_compile_definitions(dsp-citrinet-c2c PRIVATE CN_PROFILE=${_CN_PROF})

# PLL target. init_test() programs the PLL to (freq / SYS_CLK_FREQ) x 50 MHz and derives the UART
# divisor from it. NOTE: the KWS demos, which are what proved this link on silicon, ran at 500 MHz;
# citrinet and smollm both run at 750. If the link misbehaves, dropping BOTH chips to 500 MHz is the
# first thing to try — set this and SMOLLM_C2C_TARGET_FREQ_HZ together.
set(CN_C2C_TARGET_FREQ_HZ 750000000 CACHE STRING "PLL target frequency in Hz")
target_compile_definitions(dsp-citrinet-c2c PRIVATE
  DSP_CITRINET_TARGET_FREQ_HZ=${CN_C2C_TARGET_FREQ_HZ}ULL
  CN_TARGET_FREQ_HZ=${CN_C2C_TARGET_FREQ_HZ}ULL)

# Two-microphone array on the L and R slots of ONE I2S channel. See citrinet_config.h for the
# wiring and the combine modes; an unwired second mic falls back to mono at boot rather than hanging.
option(CN_C2C_MIC_STEREO "Capture a second I2S mic on the same channel's right slot" OFF)
set(_CN_STEREO 0)
if(CN_C2C_MIC_STEREO)
  set(_CN_STEREO 1)
  message(STATUS "dsp-citrinet-c2c: CN_C2C_MIC_STEREO=1 (combine=${CN_C2C_MIC_COMBINE})")
endif()
set(CN_C2C_MIC_COMBINE 1 CACHE STRING "2-mic combine: 0=left 1=average 2=difference 3=aligned")
target_compile_definitions(dsp-citrinet-c2c PRIVATE
  DSP_CITRINET_MIC_STEREO=${_CN_STEREO}
  DSP_CITRINET_MIC_COMBINE=${CN_C2C_MIC_COMBINE})

# model_data.S resolves "model/*.bin" relative to the DIRECTORY GIVEN BY -I, which must be the
# dsp-citrinet demo dir (the blobs live there, and are not duplicated here).
#
# OBJECT_DEPENDS is REQUIRED, not a nicety: the compiler emits no depfile for `.incbin`, so without
# it a regenerated model does NOT rebuild this object and the ELF keeps the OLD weights — which
# looks exactly like a flaky chip.
set_source_files_properties(${_CN_SRC}/src/model_data.S PROPERTIES
  COMPILE_OPTIONS "-I${_CN_SRC}"
  OBJECT_DEPENDS "${_CN_SRC}/model/citrinet_256_q80.bin;${_CN_SRC}/model/citrinet_audio.bin")

# include/wake MUST come before the TinySpeech include dir. Both contain a `weights.h`, and
# tinyspeech_model.c includes it by that bare name — so whichever directory is searched first
# decides which MODEL this binary runs. Ours is the marvin detector; theirs is the 6-word keyword
# classifier the Bearly demos use. Getting this order wrong builds a binary that listens for "tree".
# (Same trick, and the same hazard, as bearly-kws-llama's include ordering.)
target_include_directories(dsp-citrinet-c2c BEFORE PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/include/wake)
target_include_directories(dsp-citrinet-c2c PRIVATE
  include
  ${_CN_SRC}/include
  ${_TS_SRC}/include
  ${CMAKE_SOURCE_DIR}/c2c-demos/common
  ${CMAKE_SOURCE_DIR}/bmark-lib)

# The top-level CMakeLists does a global add_compile_options(-O1) that lands AFTER Release's -O3,
# and the last -O wins — so without a target-level override the encoder would build at -O1.
set(CN_C2C_OPT "-O3" CACHE STRING "Optimization level (overrides the global -O1)")
target_compile_options(dsp-citrinet-c2c PRIVATE ${CN_C2C_OPT})

# STACK PLACEMENT — the same fix dsp-citrinet documents at length. dsp25-llm.ld declares DRAM as
# 2 GiB (so a big .incbin'd model fits medany's +-2 GiB reach) and then derives the stack from the
# END of that fictitious region, ~1.9 GB past the last address the chip actually backs. Overflow
# there does not trap: stores silently stop reading back, or the core hangs outright. Pin the stack
# into REAL DRAM instead; the symbols are PROVIDE()d, so --defsym wins.
# 0x8FE00000 + 5 harts * 256 KiB = 0x8FF40000, leaving 768 KiB below the 0x90000000 top.
set(CN_C2C_STACK_BASE 0x8FE00000 CACHE STRING "Stack base in real DRAM")
target_link_options(dsp-citrinet-c2c PRIVATE
  -Wl,--defsym=__stack_size=0x40000               # 256 KiB per hart
  -Wl,--defsym=__stack_start=${CN_C2C_STACK_BASE}
  -Wl,--defsym=__heap_end=${CN_C2C_STACK_BASE}    # sbrk stops below the stacks
  -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/dsp-citrinet-c2c.map
)

# Catch the next oversized frame at BUILD time rather than as a flaky chip.
target_compile_options(dsp-citrinet-c2c PRIVATE -Wstack-usage=16384)

message(STATUS "dsp-citrinet-c2c: RVV=ON tile_t=${CN_C2C_TILE_T} conv1d=OFF dualcore=OFF "
               "freq=${CN_C2C_TARGET_FREQ_HZ} opt=${CN_C2C_OPT}")

# SCALAR TinySpeech KERNELS — a correctness requirement on this chip, not a preference.
#
# The boot golden self-test proved the RVV kernels are WRONG on dsp25: fed fixed feature maps whose
# logits were computed on the host from these exact weights, all 8 cases failed, every one picked
# the SAME class with the SAME ranking regardless of input, and magnitudes were 10-100x the
# reference — the signature of convolutions producing a spatially near-uniform map. Rebuilt scalar,
# the same test passes with max_diff = 0 on every case, and live wake margins went from -88 to the
# expected few-tenths. The microphone and the MFCC front-end were never involved.
#
# The whole project supports a no-RVV mode (RVV_TYPE=0 -> -march=rv64imafd), so modules.c and friends
# carry complete scalar paths gated on `#if defined(__riscv_vector)`. Compiling ONLY the TinySpeech
# translation units without the `v` extension undefines that macro and selects them. The ABI (lp64d)
# and code model are unchanged, so this links normally — and Citrinet keeps its RVV kernels, which
# are the ones that actually matter for throughput.
#
# Cost: ~6M MACs, ~240 ms scalar, once per detected onset. That is latency on a sound you made, not
# a real-time budget: nothing is being tracked while it runs.
option(DSP_WAKE_SCALAR_KERNELS "Compile the TinySpeech kernels without RVV (see comment)" ON)
if(DSP_WAKE_SCALAR_KERNELS)
  set_source_files_properties(
    ${_TS_SRC}/src/tinyspeech_model.c
    ${_TS_SRC}/src/tinyspeech_int8.c
    ${_TS_SRC}/src/tensor.c
    ${_TS_SRC}/src/misc.c
    ${_TS_SRC}/src/modules.c
    TARGET_DIRECTORY dsp-citrinet-c2c
    PROPERTIES COMPILE_OPTIONS "-march=rv64imafd")
  message(STATUS "dsp-citrinet-c2c: TinySpeech kernels SCALAR (-march=rv64imafd) — "
                 "the RVV path fails the golden self-test on dsp25")
else()
  message(WARNING "dsp-citrinet-c2c: TinySpeech kernels built with RVV — these are KNOWN WRONG on "
                  "dsp25 and the boot golden self-test will report FAIL")
endif()

# FLOAT TinySpeech pipeline, stated explicitly rather than inherited from tinyspeech_model.c's
# #ifndef default. The int8 path's conv2 kernel returns garbage on this silicon (the same RVV
# integer-reduction defect family), so which pipeline this builds is a correctness question.
#
# Boot self-test: run the model on fixed feature maps whose logits were computed on the host from
# THESE weights (include/wake/wake_golden.h, from scripts/gen_wake_golden.py). It is the bisect for
# "the wake margin looks wrong" — mic, MFCC and quantization are all bypassed, so a mismatch means
# the kernels and a match means the front-end. Costs 8 inferences at boot. Leave it ON: it is what
# would catch someone flipping DSP_WAKE_SCALAR_KERNELS off.
option(DSP_WAKE_GOLDEN_CHECK "Run the golden inference self-test at boot" ON)
set(_WAKE_GOLD 0)
if(DSP_WAKE_GOLDEN_CHECK)
  set(_WAKE_GOLD 1)
endif()
target_compile_definitions(dsp-citrinet-c2c PRIVATE
  TINYSPEECH_INT8_PIPELINE=0
  DSP_WAKE_GOLDEN_CHECK=${_WAKE_GOLD})

# COMMAND-CAPTURE onset, retuned for the wake-word flow. dsp-citrinet standalone is unaffected.
#
# The stock 1.5e-3 is calibrated for "walk up to a silent chip and say something deliberately". Here
# the capture starts immediately after the wake word, mid-conversation, at whatever level the
# question happens to be spoken — and 1.5e-3 is 7.5x the wake gate's own 2e-4. The result is that
# "marvin" trips the gate at normal volume and then the capture waits for something much louder,
# starting PART-WAY INTO the first word: "hello how are you" transcribed as "hall how are you", and
# one utterance was cut to "he". The rest of each sentence came through fine, which is what says this
# is clipping and not the acoustic model.
#
# Matching the two thresholds (slightly above the gate's, so the capture is not triggered by the room
# while the gate is deliberately twitchy) and doubling the pre-roll to 400 ms covers both the level
# mismatch and the ~240 ms the mic goes unread during the scalar inference.
set(CN_C2C_VAD_THRESHOLD "3.0e-4f" CACHE STRING "Command-capture onset (stock standalone is 1.5e-3)")
set(CN_C2C_VAD_PREROLL_SAMPLES 6400 CACHE STRING "Audio kept from before the command onset (400 ms)")
target_compile_definitions(dsp-citrinet-c2c PRIVATE
  DSP_CITRINET_VAD_THRESHOLD=${CN_C2C_VAD_THRESHOLD}
  DSP_CITRINET_VAD_PREROLL_SAMPLES=${CN_C2C_VAD_PREROLL_SAMPLES}u)
message(STATUS "dsp-citrinet-c2c: command capture onset>${CN_C2C_VAD_THRESHOLD} "
               "preroll=${CN_C2C_VAD_PREROLL_SAMPLES} samples")

# Wake-gate tunables (see include/wake_config.h for what each one costs and buys). The two that
# decide whether it fires when you speak and stays quiet when you do not are the onset threshold and
# the margin; tune both from the console, which logs the idle energy and every onset's margin.
set(DSP_WAKE_MARGIN "2.0f" CACHE STRING "Wake margin: logit[marvin] - best other logit")
set(DSP_WAKE_VAD_THRESHOLD "2.0e-4f" CACHE STRING "AC energy that counts as an onset")
set(DSP_WAKE_PREROLL_SAMPLES 3200 CACHE STRING "Audio kept from before the onset (3200 = 200 ms)")
set(DSP_WAKE_LOG_EVERY 250 CACHE STRING "Log idle energy every N 20 ms frames (0 = never)")
set(DSP_WAKE_DEBUG_FEATURES 0 CACHE STRING "Dump raw logits every Nth onset (0 = off)")
target_compile_definitions(dsp-citrinet-c2c PRIVATE
  DSP_WAKE_MARGIN=${DSP_WAKE_MARGIN}
  DSP_WAKE_VAD_THRESHOLD=${DSP_WAKE_VAD_THRESHOLD}
  DSP_WAKE_PREROLL_SAMPLES=${DSP_WAKE_PREROLL_SAMPLES}u
  DSP_WAKE_LOG_EVERY=${DSP_WAKE_LOG_EVERY}u
  DSP_WAKE_DEBUG_FEATURES=${DSP_WAKE_DEBUG_FEATURES}u)
message(STATUS "dsp-citrinet-c2c: wake word `marvin` onset>${DSP_WAKE_VAD_THRESHOLD} "
               "margin>${DSP_WAKE_MARGIN} preroll=${DSP_WAKE_PREROLL_SAMPLES}")

# glossy (whole-archive) provides the dsp25 HAL (UART, I2S); m provides cosf/sinf/expf/logf/sqrtf.
# mfcclib is the MFCC front-end the wake gate runs per window.
target_link_libraries(dsp-citrinet-c2c PRIVATE
  mfcclib
  -L${CMAKE_BINARY_DIR}/glossy -Wl,--whole-archive glossy -Wl,--no-whole-archive
  m
)
