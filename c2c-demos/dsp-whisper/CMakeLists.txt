# dsp-whisper: Whisper tiny.en on DSP 25 (plan 005). Current build = P0 log-mel FRONT-END SCAFFOLD.
# Standalone (no C2C link, no I2S). Build with the DSP toolchain:
#   make build CHIP=dsp25 TARGET=dsp-whisper
# It computes the Whisper log-mel in C on an embedded golden test vector and prints FRONTEND-CMP +
# PASS/FAIL. Regenerate the embedded tables/testvec with:
#   dsp25-tests/whisper-test/scripts/gen_whisper_frontend.py  (writes include/whisper_frontend_*.h)

add_executable(dsp-whisper
  src/main.c
  src/whisper_frontend.c
  src/whisper.c
  src/whisper_rvv.S
  src/model_data.S
  ${CMAKE_SOURCE_DIR}/bmark-lib/simple_setup.c   # init_test / PLL setup (used by the mic mode)
  ${CMAKE_SOURCE_DIR}/thread-lib/hthread.c       # 2-hart dispatch (used by the dual-core matmul)
)

# .incbin in model_data.S resolves paths relative to the demo dir (the model/ subdir holds the
# ~44 MB Q8_0 weights + golden testcase, generated by the whisper-test scripts).
set_source_files_properties(src/model_data.S PROPERTIES
  COMPILE_OPTIONS "-I${CMAKE_CURRENT_SOURCE_DIR}"
  OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/model/whisper_tiny_en_q80.bin;${CMAKE_CURRENT_SOURCE_DIR}/model/whisper_testcase.bin"
)

# The engine malloc()s a few MiB of activation buffers per encode/decode (peak ~5-8 MiB for the
# trim/mic demos). The linker default heap is only 1 MiB (malloc would return NULL -> a fault that
# spins the trap handler). NOTE: keep this modest — uart_tsi uploads the whole segment MemSiz over
# UART, zero-filling the heap, so an oversized heap makes flashing take *minutes*. 24 MiB is safe
# headroom here; a full 30 s clip (3000 frames) would need ~48 MiB — raise it only for that.
target_link_options(dsp-whisper PRIVATE
  -Wl,--defsym=__heap_size=0x1800000    # 24 MiB (was 128 MiB -> cut ~104 MiB of zero-fill upload)
  -Wl,--defsym=__stack_size=0x10000     # 64 KiB
)

# Vectorize the hot matvec kernel with RISC-V V (VLEN=256). OFF = scalar reference oracle.
option(WHISPER_USE_RVV "Use RVV-vectorized matvec_q8" OFF)
set(WHISPER_USE_RVV_VALUE 0)
if(WHISPER_USE_RVV)
  set(WHISPER_USE_RVV_VALUE 1)
  message(STATUS "dsp-whisper: WHISPER_USE_RVV=1 (RVV matvec)")
endif()
target_compile_definitions(dsp-whisper PRIVATE WHISPER_USE_RVV=${WHISPER_USE_RVV_VALUE})

# Quantize activations to int8 (int8xint8->int32 matvec). Implies the RVV asm kernel on-device.
option(WHISPER_INT8_ACT "Quantize activations to int8" OFF)
set(WHISPER_INT8_ACT_VALUE 0)
if(WHISPER_INT8_ACT)
  set(WHISPER_INT8_ACT_VALUE 1)
  message(STATUS "dsp-whisper: WHISPER_INT8_ACT=1 (int8 activations)")
endif()
target_compile_definitions(dsp-whisper PRIVATE WHISPER_INT8_ACT=${WHISPER_INT8_ACT_VALUE})

# Coarse per-kernel cycle profiling (rdcycle around matmul/attn/conv/gelu/layernorm/quant).
option(WHISPER_PROFILE "Per-kernel cycle profiling" OFF)
set(WHISPER_PROFILE_VALUE 0)
if(WHISPER_PROFILE)
  set(WHISPER_PROFILE_VALUE 1)
  message(STATUS "dsp-whisper: WHISPER_PROFILE=1")
endif()
target_compile_definitions(dsp-whisper PRIVATE WHISPER_PROFILE=${WHISPER_PROFILE_VALUE})

# Outer-product RVV front-end DFT (no reductions -> silicon-safe, unlike the old vfredusum version).
option(WHISPER_FE_RVV "RVV outer-product front-end DFT" OFF)
set(WHISPER_FE_RVV_VALUE 0)
if(WHISPER_FE_RVV)
  set(WHISPER_FE_RVV_VALUE 1)
  message(STATUS "dsp-whisper: WHISPER_FE_RVV=1 (RVV front-end DFT)")
endif()
target_compile_definitions(dsp-whisper PRIVATE WHISPER_FE_RVV=${WHISPER_FE_RVV_VALUE})

# Register-tiled outer-product GEMM (vec-nn/XNNPACK style): K-major weights streamed sequentially,
# one weight vector reused across MR output positions in vector-register accumulators. Cuts encoder
# weight DRAM traffic on silicon. Requires WHISPER_USE_RVV. Weights transposed to K-major at first use.
option(WHISPER_TILED "Register-tiled K-major GEMM" OFF)
set(WHISPER_TILED_VALUE 0)
if(WHISPER_TILED)
  set(WHISPER_TILED_VALUE 1)
  message(STATUS "dsp-whisper: WHISPER_TILED=1 (register-tiled GEMM)")
endif()
target_compile_definitions(dsp-whisper PRIVATE WHISPER_TILED=${WHISPER_TILED_VALUE})

# Split the matmul across 2 harts (requires WHISPER_USE_RVV). Run under `spike -p2` / real 2-hart HW.
option(WHISPER_DUALCORE "Split matmul across 2 harts" OFF)
set(WHISPER_DUALCORE_VALUE 0)
if(WHISPER_DUALCORE)
  set(WHISPER_DUALCORE_VALUE 1)

endif()
target_compile_definitions(dsp-whisper PRIVATE WHISPER_DUALCORE=${WHISPER_DUALCORE_VALUE})

option(WHISPER_DUALCORE_DEBUG "Print around the first dual-core fork/join" OFF)
set(WHISPER_DUALCORE_DEBUG_VALUE 0)
if(WHISPER_DUALCORE_DEBUG)
  set(WHISPER_DUALCORE_DEBUG_VALUE 1)
endif()
target_compile_definitions(dsp-whisper PRIVATE WHISPER_DUALCORE_DEBUG=${WHISPER_DUALCORE_DEBUG_VALUE})

# Run modes. FROM_AUDIO transcribes the embedded audio blob (Spike-testable); USE_MIC captures live
# I2S audio (real silicon only, build PLATFORM=CHIP). Default = validate against the golden mel.
option(DSP_WHISPER_FROM_AUDIO "Transcribe the embedded audio blob" OFF)
option(DSP_WHISPER_USE_MIC "Capture + transcribe live I2S mic audio (CHIP only)" OFF)
set(DSP_WHISPER_FROM_AUDIO_VALUE 0)
set(DSP_WHISPER_USE_MIC_VALUE 0)
if(DSP_WHISPER_FROM_AUDIO)
  set(DSP_WHISPER_FROM_AUDIO_VALUE 1)
  message(STATUS "dsp-whisper: DSP_WHISPER_FROM_AUDIO=1 (embedded audio -> transcript)")
endif()
if(DSP_WHISPER_USE_MIC)
  set(DSP_WHISPER_USE_MIC_VALUE 1)
  message(STATUS "dsp-whisper: DSP_WHISPER_USE_MIC=1 (live I2S mic)")
endif()
target_compile_definitions(dsp-whisper PRIVATE
  DSP_WHISPER_FROM_AUDIO=${DSP_WHISPER_FROM_AUDIO_VALUE}
  DSP_WHISPER_USE_MIC=${DSP_WHISPER_USE_MIC_VALUE}
)

target_include_directories(dsp-whisper PUBLIC include)
target_include_directories(dsp-whisper PRIVATE
  ${CMAKE_SOURCE_DIR}/thread-lib   # must precede bmark-lib: use thread-lib's hthread.h (has init/barrier)
  ${CMAKE_SOURCE_DIR}/bmark-lib
)

# glossy (whole-archive) provides the dsp25 HAL (UART etc.), same as the other dsp-* demos.
# m provides log10f / cos / sin used by the front-end.
target_link_libraries(dsp-whisper PRIVATE
  -L${CMAKE_BINARY_DIR}/glossy -Wl,--whole-archive glossy -Wl,--no-whole-archive
  m
)
